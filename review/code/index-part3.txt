function applyMuteState() {
  const tip = $('chat-mute-tip'), bar = $('chat-input-bar'), inp = $('chat-input'), send = $('btn-send');
  if (!S.conv || !isGroupKey(S.conv)) {
    if (tip) tip.classList.remove('show');
    if (bar) bar.classList.remove('muted');
    if (inp) { inp.disabled = false; inp.placeholder = '说点什么…'; }
    if (send) send.disabled = false;
    return;
  }
  const st = groupMuteState(gidOf(S.conv));
  if (!st.muted) {
    if (bar) bar.classList.remove('muted');
    if (inp) { inp.disabled = false; inp.placeholder = '说点什么…'; }
    if (send) send.disabled = false;
    if (tip) {
      if (st.all) { tip.textContent = '全员禁言中（群主/管理员仍可发言）'; tip.classList.add('show'); }
      else tip.classList.remove('show');
    }
    return;
  }
  const d = new Date(st.until || now());
  const when = (d.getMonth() + 1) + '月' + d.getDate() + '日 ' + pad2(d.getHours()) + ':' + pad2(d.getMinutes());
  if (tip) { tip.textContent = st.all ? '全员禁言中，只有群主和管理员可以发言' : '你被禁言到 ' + when + '，暂时不能发言'; tip.classList.add('show'); }
  if (bar) bar.classList.add('muted');
  if (inp) { inp.disabled = true; inp.placeholder = '你已被禁言'; }
  if (send) send.disabled = true;
}

    if (error) throw error;

      const m = rowToMsg(r), t = Date.parse(m.created_at) || 0;

      const cur = GRP.cursor[m.conv] || 0;

    });

  } catch (e) { logErr('grpPoll', e); }

function fmtListTime(t) {
  const d = new Date(t), n = new Date(); if (isNaN(d.getTime())) return '';
  const diff = (n - d) / 1000;
  if (diff < 60) return '刚刚';
  if (d.toDateString() === n.toDateString()) return pad2(d.getHours()) + ':' + pad2(d.getMinutes());
  if (d.toDateString() === new Date(n.getTime() - 864e5).toDateString()) return '昨天';
  return fmtTime(t);
}
function fmtMomentTime(t) {
  const d = new Date(t), n = new Date(); if (isNaN(d.getTime())) return '';
  const diff = (n - d) / 1000;
  if (diff < 60) return '刚刚';
  if (diff < 3600) return Math.floor(diff / 60) + ' 分钟前';
  const hm = pad2(d.getHours()) + ':' + pad2(d.getMinutes());
  if (d.toDateString() === n.toDateString()) return '今天 ' + hm;
  if (d.toDateString() === new Date(n.getTime() - 864e5).toDateString()) return '昨天 ' + hm;
  if (d.getFullYear() === n.getFullYear()) return (d.getMonth() + 1) + '月' + d.getDate() + '日';
  return d.getFullYear() + '年' + (d.getMonth() + 1) + '月' + d.getDate() + '日';
}

/* 读视频时长，并截第 0.1 秒的画面做封面（最长边 720 像素的 JPEG）。只用手机里的文件，不联网；视频本身不动 */
function videoProbe(file) {
  return new Promise(res => {
    let done = false, u = '', dur = null;
    const v = document.createElement('video');
    const fin = poster => {
      if (done) return; done = true; clearTimeout(t);
      try { v.removeAttribute('src'); v.load(); } catch (e) {} try { URL.revokeObjectURL(u); } catch (e) {}
      res({ dur, poster: poster || null });
    };
    const t = setTimeout(() => fin(null), 6000);
    const grab = () => {
      try {
        const w = v.videoWidth, h = v.videoHeight; if (!w || !h) { fin(null); return; }
        const k = Math.min(1, 720 / Math.max(w, h)), c = document.createElement('canvas');
        c.width = Math.max(1, Math.round(w * k)); c.height = Math.max(1, Math.round(h * k));
        const g = c.getContext('2d'); g.drawImage(v, 0, 0, c.width, c.height);
        const px = g.getImageData(0, 0, c.width, c.height).data; let lit = 0;   // 截到的是全黑（有些手机没解出画面）就不用
        for (let i = 0; i < px.length; i += 4 * 97) if (px[i] + px[i + 1] + px[i + 2] > 24) { lit = 1; break; }
        if (!lit || !c.toBlob) { fin(null); return; }
        c.toBlob(b => fin(b && b.size > 0 ? b : null), 'image/jpeg', 0.82);
      } catch (e) { fin(null); }
    };
    v.muted = true; v.playsInline = true; v.setAttribute('playsinline', ''); v.setAttribute('webkit-playsinline', ''); v.preload = 'auto';
    v.onloadedmetadata = () => { const d = v.duration; dur = d > 0 && isFinite(d) ? Math.min(86400, Math.round(d)) : null; try { v.currentTime = Math.min(0.1, (d || 1) / 2); } catch (e) { fin(null); } };
    v.onseeked = () => { v.onseeked = null; grab(); };
    v.onerror = () => fin(null);
    try { u = URL.createObjectURL(file); v.src = u; v.load(); } catch (e) { fin(null); }
  });
}

async function vpvEncode() {
  const v = $('vpv-video');
  if (!v.captureStream) throw new Error('这台设备的浏览器不支持视频编辑，只能原样发送');
  const start = Math.max(0, VPV.start), end = Math.min(VPV.end, VPV.dur || v.duration || 0);
  if (end - start < 0.2) throw new Error('裁剪后太短了');
  const w = v.videoWidth || 720, h = v.videoHeight || 1280;
  const scale = VPV.compress ? Math.min(1, 720 / Math.max(w, h)) : 1;
  const cw = Math.max(2, Math.round(w * scale / 2) * 2), ch = Math.max(2, Math.round(h * scale / 2) * 2);
  const cv = document.createElement('canvas'); cv.width = cw; cv.height = ch;
  const ctx = cv.getContext('2d');
  const mixed = new MediaStream();
  cv.captureStream(30).getVideoTracks().forEach(function (t) { mixed.addTrack(t); });
  try { const vs = v.captureStream(); if (vs) vs.getAudioTracks().forEach(function (t) { mixed.addTrack(t); }); } catch (e) {}
  const mime = (window.MediaRecorder && MediaRecorder.isTypeSupported && MediaRecorder.isTypeSupported('video/webm;codecs=vp8,opus')) ? 'video/webm;codecs=vp8,opus' : 'video/webm';
  const rec = new MediaRecorder(mixed, { mimeType: mime, videoBitsPerSecond: VPV.compress ? 1600000 : 4000000, audioBitsPerSecond: 96000 });
  VPV.rec = rec; VPV.cancelEncode = false;
  const parts = [];
  rec.ondataavailable = function (e) { if (e.data && e.data.size) parts.push(e.data); };
  const stopped = new Promise(function (r) { rec.onstop = r; });
  $('vpv-progress').style.display = 'flex';
  try { v.pause(); v.currentTime = start; } catch (e) {}
  await new Promise(function (r) { const h2 = function () { v.removeEventListener('seeked', h2); r(); }; v.addEventListener('seeked', h2, { once: true }); setTimeout(r, 700); });
  rec.start(120);
  try { await v.play(); } catch (e) {}
  const total = Math.max(0.2, end - start);
  await new Promise(function (resolve, reject) {
    const tick = function () {

/* ---- 文字里的网址变成可以点的链接 ---- */
const URL_RE = /(https?:\/\/[^\s<>"'，。！？、；：（）【】《》「」]+|www\.[a-z0-9-]+\.[^\s<>"'，。！？、；：（）【】《》「」]+)/gi;
function linkify(t) {
  t = String(t == null ? '' : t); let out = '', last = 0;
  String(t).replace(URL_RE, (m, g, off) => {
    const u = m.replace(/[.,!?;:)\]}>]+$/, ''); if (!u) return m;
    out += esc(t.slice(last, off)) + '<a href="' + esc(/^www\./i.test(u) ? 'https://' + u : u) + '" target="_blank" rel="noopener noreferrer">' + esc(u) + '</a>';
    last = off + u.length; return m;
  });
  return out + esc(t.slice(last));
}
function mentionsOf(m) {

  return Array.isArray(a) ? a.filter(function (x) { return x && x.acc && x.name; }) : [];
}

    if (!x || !x.name) return;
    const tok = esc('@' + x.name);
    h = h.split(tok).join('<span class="at">' + tok + '</span>');
  });
  return h;
}

/* ---- 上传进度：直接调用 Storage 接口（格式和官方库对文件的做法一样）才能拿到进度；出问题就退回官方库上传 ---- */
function xhrUpload(path, blob, ct, tok, onPct) {
  return new Promise((res, rej) => {
    const x = new XMLHttpRequest();
    x.open('POST', API_URL + '/storage/v1/object/chat-media/' + path.split('/').map(encodeURIComponent).join('/'));
    x.setRequestHeader('Authorization', 'Bearer ' + tok); x.setRequestHeader('apikey', SUPABASE_KEY); x.setRequestHeader('x-upsert', 'false');
    if (x.upload) x.upload.onprogress = e => { if (e.lengthComputable && e.total > 0) onPct(e.loaded / e.total); };
    x.onload = () => {
      if (x.status >= 200 && x.status < 300) { res(); return; }
      let j = {}; try { j = JSON.parse(x.responseText || '{}'); } catch (e) {}
      const err = new Error(j.message || j.error || ('HTTP ' + x.status)); err.status = x.status; err.statusCode = String(j.statusCode || x.status); rej(err);
    };
    x.onerror = () => { const e = new Error('Failed to fetch'); e.status = 0; rej(e); };
    x.onabort = x.onerror;
    const fd = new FormData(); fd.append('cacheControl', '31536000');
    fd.append('', ct && blob.type !== ct ? new Blob([blob], { type: ct }) : blob);
    x.send(fd);
  });
}

function u8(b64) {
  const s = String(b64 || '').replace(/-/g, '+').replace(/_/g, '/'), pad = (4 - s.length % 4) % 4, bin = atob(s + '='.repeat(pad));
  const out = new Uint8Array(bin.length); for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i); return out;
}
function sameKey(sub, pub) {
  try { const k = sub.options && sub.options.applicationServerKey; if (!k) return true; const a = new Uint8Array(k), b = u8(pub); return a.length === b.length && a.every((x, i) => x === b[i]); } catch (e) { return true; }
}

async function fnCall(body) {

const pushBlocked = () => S.v8 === false || (PUSH.srv === false && now() - PUSH.srvAt < 10 * 60000);
async function firePush(kind, to, ref) {   // 请推送函数给 to 推一条（对方在线、不是好友等情况由数据库判断后跳过）
  if (!S.me || pushBlocked()) return null;
  try {
    const j = await fnCall({ op: 'push', kind, to: to || null, ref: ref || null });
    PUSH.last = kind + (j.skip ? '：没推（' + ({ online: '对方在线', nosub: '对方没开通知', dup: '已经推过', ref: '找不到这条', friend: '不是好友', member: '不是群成员', type: '这种消息不推', to: '目标不对', auth: '登录失效' }[j.skip] || j.skip) + '）' : '：已推送到 ' + (j.sent || 0) + ' 台手机');
    return j;
  } catch (e) {
    if (e.status === 404 || e.net) { PUSH.srv = false; PUSH.srvAt = now() - (e.net ? 9 * 60000 : 0); }   // 没部署：10 分钟内不再试；网络不通：1 分钟后再试

function pushAfterSend(m) {   // 消息发成功后：对方不在线才请求推送；连着发好几条（比如一次 9 张图）只推最后一次，通知里显示“[9条]”
  if (!m || m.type === 'call' || m.type === 'recalled' || pushBlocked()) return;
  const gp = isGroupKey(m.conv);
  if (!gp && isOnline(m.to_acc)) return;

  const q = PUSH.q[key] || (PUSH.q[key] = {});
  q.kind = gp ? 'gmsg' : 'msg'; q.to = key; q.ref = m.id;
  clearTimeout(q.t);
  q.t = setTimeout(() => { delete PUSH.q[key]; firePush(q.kind, q.to, q.ref); }, 1200);
}

/* ---- 点通知打开 App：#c=账号 打开和他的聊天；#call=账号 打开聊天并马上取来电信号；#req 打开好友请求 ---- */
let linkWait = '';
function goLink(u) {
  let h = ''; try { h = new URL(u, location.href).hash; } catch (e) { h = String(u || ''); }
  const gm = /^#g=([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})$/i.exec(h);
  if (gm) {
    if (!S.started || !S.me) { linkWait = h; return; }
    linkWait = '';
    const k = 'g:' + gm[1].toLowerCase();
    grpLoad(true).then(function () { openConv(k, k); }, function () { openConv(k, k); });
    return;
  }
  if (/^#inv$/.test(h)) {
    if (!S.started || !S.me) { linkWait = h; return; }
    linkWait = '';
    goTab('requests'); grpInvitesLoad(true);
    return;
  }
  const m = /^#(c|call)=([a-z0-9_-]{1,40})$/.exec(h) || /^#(req)$/.exec(h);
  if (!m) return;
  if (!S.started || !S.me) { linkWait = h; return; }
  linkWait = '';
  if (m[1] === 'req') { goTab('requests'); return; }
  openChatWith(m[2]);
  if (m[1] === 'call' && sb && !S.noSigDb) pollSignals().catch(() => {});
}
function bindSW() {
  const h = location.hash;
  if (/^#(c=|call=|req$|inv$|g=)/.test(h)) { linkWait = h; try { history.replaceState(history.state, '', location.pathname + location.search); } catch (e) {} }
  if (!('serviceWorker' in navigator)) return;
  try {
    navigator.serviceWorker.addEventListener('message', e => {
      const d = e.data || {};
      if (d.t === 'open') goLink(d.url);
      else if (d.t === 'resub') pushSync();
      else if (d.t === 'pushed' && !document.hidden && d.from && S.me && S.conv === convKeyOf(ME(), d.from)) closeNotes(t => t === d.tag);   // 正看着这个聊天：推送来的通知直接收起
      else if (d.t === 'pushed' && !document.hidden && d.kind === 'gmsg' && d.tag && S.conv === d.tag) closeNotes(t => t === d.tag);            // 正看着这个群：群通知收起
    });
  } catch (e) {}
}

let viewportApply = null;
function bindViewport() {
  const vv = window.visualViewport; if (!vv) return;
  const desktop = window.matchMedia('(min-width: 768px) and (hover: hover) and (pointer: fine)');
  const apply = () => {
    const root = document.documentElement;
    // 只有真正的鼠标桌面才切成双栏；平板/触屏即使宽度够，也按手机的可视区处理。
    if (desktop.matches) { root.style.removeProperty('--vvh'); root.style.removeProperty('--vvt'); document.body.classList.remove('kb'); return; }
    if (!vv.height) return;
    // 聊天页始终贴合“看得见的那块屏幕”：键盘、浏览器地址栏、下拉都算在内。
    root.style.setProperty('--vvh', vv.height + 'px');
    root.style.setProperty('--vvt', Math.max(0, vv.offsetTop || 0) + 'px');
    const kb = (window.innerHeight - vv.height > 100) || (vv.height < (window.screen.height || 0) * 0.68);
    document.body.classList.toggle('kb', kb);
    if (S.conv && S.stick) requestAnimationFrame(function () { const b = $('chat-messages'); if (b) b.scrollTop = b.scrollHeight; });
  };
  viewportApply = apply;
  const resync = () => { [0, 120, 420, 1000].forEach(function (ms) { setTimeout(function () { try { apply(); } catch (e) {} }, ms); }); };
  vv.addEventListener('resize', apply); vv.addEventListener('scroll', apply); window.addEventListener('resize', apply); window.addEventListener('orientationchange', function () { setTimeout(apply, 260); });
  document.addEventListener('visibilitychange', function () { if (!document.hidden) resync(); });
  window.addEventListener('pageshow', resync); window.addEventListener('focus', resync);
  apply();
}
