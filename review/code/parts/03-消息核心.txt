const IDB = { p: null, bad: false };
function idbOpen() {
  if (IDB.p) return IDB.p;
  IDB.p = new Promise(function (res, rej) {
    if (!window.indexedDB) { IDB.bad = true; rej(new Error('no indexeddb')); return; }
    let r = null;
    try { r = indexedDB.open('chatter', 1); } catch (e) { IDB.bad = true; rej(e); return; }
    r.onupgradeneeded = function () { const db = r.result; if (!db.objectStoreNames.contains('msgs')) db.createObjectStore('msgs'); };
    r.onsuccess = function () { res(r.result); };
    r.onerror = function () { IDB.bad = true; rej(r.error || new Error('idb')); };
  });
  return IDB.p;
}
async function idbGet(k) {
  try { const db = await idbOpen(); return await new Promise(function (res, rej) { const q = db.transaction('msgs', 'readonly').objectStore('msgs').get(k); q.onsuccess = function () { res(q.result); }; q.onerror = function () { rej(q.error); }; }); }
  catch (e) { IDB.bad = true; return null; }
}
async function idbSet(k, v) {
  try {
    const db = await idbOpen();
    await new Promise(function (res, rej) { const tx = db.transaction('msgs', 'readwrite'); tx.objectStore('msgs').put(v, k); tx.oncomplete = function () { res(); }; tx.onerror = function () { rej(tx.error); }; });
  } catch (e) { IDB.bad = true; throw e; }
}
async function idbDel(k) {
  try {
    const db = await idbOpen();
    await new Promise(function (res, rej) { const tx = db.transaction('msgs', 'readwrite'); tx.objectStore('msgs').delete(k); tx.oncomplete = function () { res(); }; tx.onerror = function () { rej(tx.error); }; });
  } catch (e) { IDB.bad = true; }
}
async function idbClear() {
  try {
    const db = await idbOpen();

  } catch (e) { IDB.bad = true; }
}

async function msgCacheGet(conv) {
  if (!S.uid || !conv) return [];
  if (MSG.uid !== S.uid) { MSG.mem.clear(); MSG.uid = S.uid; }
  const k = msgKey(conv);
  if (MSG.mem.has(k)) return MSG.mem.get(k);
  let list = IDB.bad ? null : await idbGet(k);
  if (!Array.isArray(list)) {
    list = LS.get(mk(S.uid, conv), []) || [];          // 旧版 localStorage 缓存：读一次就搬过去
    if (list.length && !IDB.bad) idbSet(k, list).catch(function () {});
    if (list.length) LS.del(mk(S.uid, conv));
  }
  MSG.mem.set(k, list || []);
  return list || [];
}
function msgCacheSave(conv, list) {
  if (!S.uid || !conv) return;
  const k = msgKey(conv);
  MSG.mem.set(k, list);
  if (IDB.bad) { try { LS.set(mk(S.uid, conv), list); } catch (e) {} return; }
  idbSet(k, list).catch(function () { try { LS.set(mk(S.uid, conv), list); } catch (e) {} });
  try {                                              // 最多留 80 个会话的缓存，太久的删掉
    let idx = LS.get('ch2mgi:' + S.uid, []) || [];
    idx = idx.filter(function (x) { return x !== conv; }); idx.push(conv);
    const drop = idx.length > 80 ? idx.splice(0, idx.length - 80) : [];
    LS.set('ch2mgi:' + S.uid, idx);
    drop.forEach(function (c) { const kk = msgKey(c); MSG.mem.delete(kk); if (!IDB.bad) idbDel(kk); });
  } catch (e) {}
}
async function msgMigrate() {
  if (!S.uid || IDB.bad) return;

    const keys = [];
    for (let i = 0; i < localStorage.length; i++) { const k = localStorage.key(i); if (k && k.indexOf(pre) === 0) keys.push(k); }
    for (const k of keys) {
      const conv = k.slice(pre.length), old = LS.get(k, []);
      if (Array.isArray(old) && old.length) await idbSet(msgKey(conv), old);
      LS.del(k);
    }
    LS.del('ch2midx:' + uid);
  } catch (e) { IDB.bad = true; }
}

    return '开始聊天吧';

function bindPull(el, fn) {
  if (!el) return;
  let sx = 0, sy = 0, pulling = false, locked = false;
  const end = function () {
    if (!pulling) return;
    const dist = pullDist;
    pulling = false; locked = false;
    if (dist > 80 && !pullBusy) fn();
    else if (!pullBusy) paintPull('', false);
  };
  el.addEventListener('touchstart', function (e) {
    if (pullBusy || el.scrollTop > 0 || e.touches.length !== 1) { pulling = false; return; }
    sx = e.touches[0].clientX; sy = e.touches[0].clientY; pulling = true; locked = false; pullDist = 0;
  }, { passive: true });
  el.addEventListener('touchmove', function (e) {
    if (!pulling || pullBusy) return;
    const t = e.touches[0], dx = t.clientX - sx, dy = t.clientY - sy;
    if (!locked) {
      if (Math.abs(dx) > 12 && Math.abs(dx) > Math.abs(dy)) { pulling = false; paintPull('', false); return; }   // 斜着/横着滑：不触发
      if (dy > 12 && dy > Math.abs(dx) * 1.6) locked = true; else return;
    }
    pullDist = Math.max(0, dy);
    if (el.scrollTop > 0) { pulling = false; locked = false; paintPull('', false); return; }
    if (e.cancelable) e.preventDefault();
    paintPull(pullDist > 80 ? '松开刷新' : '下拉刷新', true);
  }, { passive: false });
  el.addEventListener('touchend', end);
  el.addEventListener('touchcancel', function () { pulling = false; locked = false; if (!pullBusy) paintPull('', false); });
}

async function compressImage(file) {
  const t = String((file && file.type) || '');
  if (!/^image\//.test(t) || /gif|svg/i.test(t) || file.size < 250 * 1024) return file;
  try {
    let src, w, h;
    if (window.createImageBitmap) {
      const bmp = await createImageBitmap(file, { imageOrientation: 'from-image' }).catch(function () { return createImageBitmap(file); });
      src = bmp; w = bmp.width; h = bmp.height;
    } else {
      const im = await loadImgEl(file);
      src = im; w = im.naturalWidth || im.width; h = im.naturalHeight || im.height;
    }
    if (!w || !h) return file;
    const max = 1600, r = Math.min(1, max / Math.max(w, h));
    const cw = Math.max(1, Math.round(w * r)), ch = Math.max(1, Math.round(h * r));
    const cv = document.createElement('canvas'); cv.width = cw; cv.height = ch;
    const ctx = cv.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, cw, ch); ctx.drawImage(src, 0, 0, cw, ch);
    if (src.close) { try { src.close(); } catch (e) {} }
    const blob = await new Promise(function (res) { cv.toBlob(res, 'image/jpeg', 0.85); });
    if (!blob || blob.size >= file.size * 0.92) return file;
    return new File([blob], String(file.name || 'image').replace(/\.[^.]+$/, '') + '.jpg', { type: 'image/jpeg' });
  } catch (e) { logErr('img-compress', e); return file; }
}

  const el = msgElOf(id); const v = el && el.querySelector('.voice');
  if (v) { v.classList.toggle('playing', on); if (!on) { const i = v.querySelector('.vprog i'); if (i) i.style.width = '0'; } }
  const sp = $('voice-speed'); if (sp) sp.style.display = on ? 'block' : 'none';
  if (on) paintSpeed();
}
let vSeekAt = 0, vSeekEl = null;
function voiceSeek(e, b) {
  if (!audioEl || !isFinite(audioEl.duration) || !audioEl.duration) return;
  const r = b.getBoundingClientRect();
  const ratio = Math.max(0, Math.min(1, (e.clientX - r.left) / r.width));
  try { audioEl.currentTime = ratio * audioEl.duration; } catch (err) {}
  const bar = b.querySelector('.vprog i'); if (bar) bar.style.width = (ratio * 100).toFixed(1) + '%';
}
function voiceProgress(id, a) {
  const upd = function () {
    if (playingId !== id) return;
    const el = msgElOf(id), bar = el && el.querySelector('.vprog i');
    if (bar && isFinite(a.duration) && a.duration) bar.style.width = (a.currentTime / a.duration * 100).toFixed(1) + '%';
  };
  a.addEventListener('timeupdate', upd);
  return upd;
}
function togglePlay(id, url) {
  if (playingId === id) { stopAudio(); return; }
  stopAudio();
  const src = disp(url);
  const a = new Audio(); audioEl = a; playingId = id; a.preload = 'auto'; a.src = src; a.playbackRate = vspeed(); markPlaying(id, true);
  voiceProgress(id, a);
  const end = () => { if (playingId === id) { playingId = null; audioEl = null; markPlaying(id, false); } };
  a.onended = end;
  let tried = false;
  // 直接播放失败时：先把文件下载成 Blob 再播（绕开服务器返回的类型/分段请求问题），仍失败就说明具体原因
  a.onerror = async () => {
    const code = a.error ? a.error.code : 0;
    logErr('voice-play', 'MediaError ' + code + ' ' + src.slice(0, 120));
    if (tried || playingId !== id || src.indexOf('blob:') === 0) { end(); toast('语音播放失败（错误码 ' + code + '）'); return; }
    tried = true;
    try {
      const r = await fetch(src, { cache: 'no-store' });
      if (!r.ok) { end(); toast('语音文件读取失败（HTTP ' + r.status + '）'); logErr('voice-http', r.status + ' ' + src.slice(0, 120)); return; }
      const buf = await r.arrayBuffer();
      if (playingId !== id) return;
      const typ = /\.m4a(\?|$)/i.test(src) || /\.mp4(\?|$)/i.test(src) ? 'audio/mp4' : /\.webm(\?|$)/i.test(src) ? 'audio/webm' : 'audio/wav';
      const b = new Blob([buf], { type: typ }), u = URL.createObjectURL(b);
      const a2 = new Audio(); audioEl = a2; a2.preload = 'auto'; a2.src = u; a2.playbackRate = vspeed();
      voiceProgress(id, a2);
      a2.onended = () => { end(); try { URL.revokeObjectURL(u); } catch (e) {} };
      a2.onerror = () => { end(); toast('语音无法解码（错误码 ' + (a2.error ? a2.error.code : 0) + '，大小 ' + buf.byteLength + '）'); };
      const p2 = a2.play(); if (p2 && p2.catch) p2.catch(e => { if (e && e.name === 'NotAllowedError') { end(); toast('无法播放语音，请再点一次'); } });
    } catch (e) { end(); toast('语音下载失败：网络连不上存储服务'); logErr('voice-fetch', e); }
  };
  const p = a.play(); if (p && p.catch) p.catch(e => { if (e && e.name === 'NotAllowedError') { end(); toast('无法播放语音'); } });
}

/* ================= 录音（优先 AAC/M4A，失败回退 WAV；通话不受影响）================= */
const rec = { state: 'idle', token: 0, ctx: null, stream: null, src: null, proc: null, chunks: [], len: 0, rate: 44100, t0: 0, y0: 0, cancel: false, tm: 0, mr: null, mrChunks: [], mime: '' };
function pickVoiceMime() {
  if (!window.MediaRecorder || !window.MediaRecorder.isTypeSupported) return '';
  const list = ['audio/mp4;codecs=mp4a.40.2', 'audio/mp4'];   // AAC：iPhone/安卓/电脑都能放
  for (const m of list) { try { if (MediaRecorder.isTypeSupported(m)) return m; } catch (e) {} }
  return '';
}
function recProblem() {
  if (iosAppWebview()) return IOS_APP_MSG;
  if (!window.isSecureContext) return '录音需要 https 地址（或 localhost）';
  if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) return '当前浏览器不支持录音（微信/QQ 等内置浏览器常见），请用 Safari / Chrome 打开';
  if (!(window.AudioContext || window.webkitAudioContext)) return '当前浏览器不支持录音';
  return '';
}
function recUI(on, text) {
  const o = $('record-overlay'); o.classList.toggle('active', !!on); if (!on) o.classList.remove('cancel');
  if (text) $('record-text').textContent = text;
  $('btn-audio').textContent = on ? '松开 发送' : '按住 说话';
}
function recCleanup() {
  clearInterval(rec.tm);
  try { if (rec.mr && rec.mr.state !== 'inactive') rec.mr.stop(); } catch (e) {}
  try { if (rec.proc) { rec.proc.onaudioprocess = null; rec.proc.disconnect(); } if (rec.src) rec.src.disconnect(); } catch (e) {}
  stopStream(rec.stream);
  try { if (rec.ctx && rec.ctx.state !== 'closed') { const p = rec.ctx.close(); if (p && p.catch) p.catch(() => {}); } } catch (e) {}
  rec.stream = rec.src = rec.proc = rec.ctx = rec.mr = null; rec.chunks = []; rec.len = 0; rec.mrChunks = []; rec.mime = ''; rec.state = 'idle';
  recUI(false);
}
function recAbort() { if (rec.state === 'idle') return; rec.token++; recCleanup(); }
function recTick() {
  const s = (now() - rec.t0) / 1000;
  $('record-overlay').classList.toggle('cancel', rec.cancel);
  $('record-text').textContent = Math.floor(s) + '″  ' + (rec.cancel ? '松开手指，取消发送' : '手指上滑，取消发送');
  if (s >= 60) recEnd(false);
}
function recStart(y) {
  if (rec.state !== 'idle' || !S.conv) return;
  const pr = recProblem(); if (pr) { toast(pr); return; }
  const token = ++rec.token; rec.state = 'starting'; rec.y0 = y || 0; rec.cancel = false; rec.chunks = []; rec.len = 0; rec.mrChunks = []; rec.mime = ''; rec.mr = null;
  recUI(true, '手指上滑，取消发送');
  let ctx;
  try {
    const AC = window.AudioContext || window.webkitAudioContext;
    ctx = new AC(); rec.ctx = ctx; rec.rate = ctx.sampleRate || 44100;
    if (ctx.state === 'suspended') ctx.resume().catch(() => {});   // iOS：必须在手势里创建/恢复
  } catch (e) { logErr('rec-ctx', e); recCleanup(); toast('无法初始化录音'); return; }
  gum({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true } }, 20000).then(stream => {
    if (token !== rec.token || rec.state !== 'starting') { stopStream(stream); return; }
    try {
      rec.stream = stream;
      const src = ctx.createMediaStreamSource(stream), proc = ctx.createScriptProcessor(4096, 1, 1);
      proc.onaudioprocess = e => { if (rec.state !== 'recording') return; const d = e.inputBuffer.getChannelData(0); rec.chunks.push(new Float32Array(d)); rec.len += d.length; };
      src.connect(proc); proc.connect(ctx.destination); rec.src = src; rec.proc = proc;
      rec.mime = pickVoiceMime();
      if (rec.mime) {
        try {
          const mr = new MediaRecorder(stream, { mimeType: rec.mime, audioBitsPerSecond: 48000 });
          mr.ondataavailable = e => { if (e.data && e.data.size) rec.mrChunks.push(e.data); };
          mr.start(250); rec.mr = mr;
        } catch (e) { logErr('mr-start', e); rec.mime = ''; rec.mr = null; }
      }
      if (ctx.state === 'suspended') ctx.resume().catch(() => {});
      rec.state = 'recording'; rec.t0 = now(); rec.tm = setInterval(recTick, 200);
    } catch (e) { logErr('rec-start', e); recCleanup(); toast('无法开始录音'); }
  }).catch(e => { if (token !== rec.token) return; logErr('rec-gum', e); recCleanup(); toast('无法录音：' + micError(e)); });
}
function recMove(y) { if (rec.state === 'idle') return; rec.cancel = rec.y0 - y > 60; }
async function recEnd(cancel) {
  if (rec.state === 'idle') return;
  if (rec.state === 'starting') { rec.token++; recCleanup(); return; }
  if (rec.state !== 'recording') return;
  const token = rec.token;
  const dc = cancel || rec.cancel, sec = (now() - rec.t0) / 1000;
  rec.state = 'stopping';
  let blob = null, ext = 'wav', ct = 'audio/wav';
  if (!dc && rec.mr) {
    blob = await new Promise(function (res) {
      let done = false;
      const fin = function () {
        if (done) return; done = true;
        const b = new Blob(rec.mrChunks, { type: (rec.mime || 'audio/mp4').split(';')[0] });
        res(b.size ? b : null);
      };
      try { rec.mr.onstop = fin; if (rec.mr.state !== 'inactive') rec.mr.stop(); else fin(); } catch (e) { fin(); }
      setTimeout(fin, 1800);
    });
    if (blob) {
      if (/mp4/i.test(rec.mime)) { ext = 'm4a'; ct = 'audio/mp4'; }
      else if (/webm/i.test(rec.mime)) { ext = 'webm'; ct = 'audio/webm'; }
    }
  }
  const wav = (!dc && !blob) ? encodeWav(rec.chunks, rec.len, rec.rate) : null;
  recCleanup();
  if (token !== rec.token) return;          // 这期间被取消/关了聊天
  if (dc) { SFX.cancel(); return; }
  if (sec < 0.8) { toast('说话时间太短'); return; }
  if (blob && blob.size) { sendVoice(blob, Math.max(1, Math.round(sec)), ext, ct); return; }
  if (wav && wav.size) { sendVoice(wav, Math.max(1, Math.round(sec)), 'wav', 'audio/wav'); return; }
  toast('录音失败，请再试一次');
}
function encodeWav(chunks, len, srcRate) {
  const target = 12000, ratio = srcRate / target, outLen = Math.max(1, Math.floor(len / ratio));
  const merged = new Float32Array(len); let o = 0, peak = 0;
  for (const c of chunks) { merged.set(c, o); o += c.length; }
  for (let i = 0; i < len; i++) { const a = Math.abs(merged[i]); if (a > peak) peak = a; }
  const gain = peak > 0.0001 ? Math.min(4, 0.9 / peak) : 1;
  const buf = new ArrayBuffer(44 + outLen * 2), v = new DataView(buf);
  const w = (off, s) => { for (let i = 0; i < s.length; i++) v.setUint8(off + i, s.charCodeAt(i)); };
  w(0, 'RIFF'); v.setUint32(4, 36 + outLen * 2, true); w(8, 'WAVE'); w(12, 'fmt '); v.setUint32(16, 16, true); v.setUint16(20, 1, true); v.setUint16(22, 1, true);
  v.setUint32(24, target, true); v.setUint32(28, target * 2, true); v.setUint16(32, 2, true); v.setUint16(34, 16, true); w(36, 'data'); v.setUint32(40, outLen * 2, true);
  for (let i = 0; i < outLen; i++) {
    const s = Math.floor(i * ratio), e = Math.min(len, Math.max(s + 1, Math.floor((i + 1) * ratio)));
    let sum = 0; for (let j = s; j < e; j++) sum += merged[j];
    let x = (sum / Math.max(1, e - s)) * gain; x = Math.max(-1, Math.min(1, x));
    v.setInt16(44 + i * 2, x < 0 ? x * 0x8000 : x * 0x7fff, true);
  }
  return new Blob([buf], { type: 'audio/wav' });
}
