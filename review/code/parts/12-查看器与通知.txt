function u8(b64) {

}

}

async function firePush(kind, to, ref) {   // 请推送函数给 to 推一条（对方在线、不是好友等情况由数据库判断后跳过）

    linkWait = '';
    const k = 'g:' + gm[1].toLowerCase();
    grpLoad(true).then(function () { openConv(k, k); }, function () { openConv(k, k); });
    return;
  }
  if (/^#inv$/.test(h)) {
    if (!S.started || !S.me) { linkWait = h; return; }
    linkWait = '';
    goTab('requests'); grpInvitesLoad(true);
    return;
  }
  const m = /^#(c|call)=([a-z0-9_-]{1,40})$/.exec(h) || /^#(req)$/.exec(h);
  if (!m) return;
  if (!S.started || !S.me) { linkWait = h; return; }
  linkWait = '';
  if (m[1] === 'req') { goTab('requests'); return; }
  openChatWith(m[2]);
  if (m[1] === 'call' && sb && !S.noSigDb) pollSignals().catch(() => {});
}
function bindSW() {
  const h = location.hash;
  if (/^#(c=|call=|req$|inv$|g=)/.test(h)) { linkWait = h; try { history.replaceState(history.state, '', location.pathname + location.search); } catch (e) {} }
  if (!('serviceWorker' in navigator)) return;
  try {
    navigator.serviceWorker.addEventListener('message', e => {
      const d = e.data || {};
      if (d.t === 'open') goLink(d.url);
      else if (d.t === 'resub') pushSync();
      else if (d.t === 'pushed' && !document.hidden && d.from && S.me && S.conv === convKeyOf(ME(), d.from)) closeNotes(t => t === d.tag);   // 正看着这个聊天：推送来的通知直接收起
      else if (d.t === 'pushed' && !document.hidden && d.kind === 'gmsg' && d.tag && S.conv === d.tag) closeNotes(t => t === d.tag);            // 正看着这个群：群通知收起
    });
  } catch (e) {}
}
