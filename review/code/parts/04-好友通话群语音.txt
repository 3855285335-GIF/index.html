}

  const now0 = Date.now(), cur = { t: now0, rx: 0, tx: 0, lost: 0, rcv: 0 };
  let inV = null, outV = null, pair = null, byId = {};
  rep.forEach(r => { byId[r.id] = r; });
  rep.forEach(r => {
    if (r.type === 'inbound-rtp') { cur.rx += r.bytesReceived || 0; cur.lost += r.packetsLost || 0; cur.rcv += r.packetsReceived || 0; if (r.kind === 'video') inV = r; }
    else if (r.type === 'outbound-rtp') { cur.tx += r.bytesSent || 0; if (r.kind === 'video') outV = r; }
    else if (r.type === 'candidate-pair' && (r.nominated || r.selected) && r.state === 'succeeded' && (!pair || (r.bytesReceived || 0) + (r.bytesSent || 0) > (pair.bytesReceived || 0) + (pair.bytesSent || 0))) pair = r;
  });
  rep.forEach(r => { if (r.type === 'transport' && r.selectedCandidatePairId && byId[r.selectedCandidatePairId]) pair = byId[r.selectedCandidatePairId]; });
  const pv = call.prevStats; call.prevStats = cur;
  let rxK = null, txK = null, lossP = null;
  if (pv && cur.t > pv.t) {
    const dt = (cur.t - pv.t) / 1000;
    rxK = Math.max(0, (cur.rx - pv.rx) * 8 / dt / 1000); txK = Math.max(0, (cur.tx - pv.tx) * 8 / dt / 1000);
    const dl = Math.max(0, cur.lost - pv.lost), dr = Math.max(0, cur.rcv - pv.rcv);
    lossP = dl + dr > 0 ? dl * 100 / (dl + dr) : 0;
  }
  const rtt = pair && pair.currentRoundTripTime != null ? Math.round(pair.currentRoundTripTime * 1000) : null;
  let route = '';
  if (pair) {
    const lc = byId[pair.localCandidateId], rc = byId[pair.remoteCandidateId];
    route = (lc && lc.candidateType === 'relay') || (rc && rc.candidateType === 'relay') ? '中转' : '直连';
  }
  if (route && now0 - (call.relayChk || 0) >= 4000) {         // 每 4 秒复查一次是否走中转
    call.relayChk = now0;
    const relay = route === '中转';
    if (relay !== !!call.relay) {
      call.relay = relay; applyCallParams(call);
      if (relay && relaySave() && !call.relayTold) { call.relayTold = true; toast('正在走中转服务器：已自动降低码率，节省中转流量'); }
    }
  }
  call.stat = { rxK, txK, lossP, rtt, route, rxRes: inV && inV.frameWidth ? inV.frameWidth + '×' + inV.frameHeight : '', rxFps: inV && inV.framesPerSecond != null ? Math.round(inV.framesPerSecond) : null, txRes: outV && outV.frameWidth ? outV.frameWidth + '×' + outV.frameHeight : '', txFps: outV && outV.framesPerSecond != null ? Math.round(outV.framesPerSecond) : null, limit: outV && outV.qualityLimitationReason && outV.qualityLimitationReason !== 'none' ? outV.qualityLimitationReason : '' };
  const lv = pv ? sigLevel(lossP, rtt) : 4;
  const sg = $('call-sig'); sg.className = 'l' + lv;

}

function spkLevel(m) {
  if (!m) return 0;
  m.an.getByteFrequencyData(m.buf);
  let s = 0; for (let i = 0; i < m.buf.length; i++) s += m.buf[i];
  return s / m.buf.length;
}
function spkStop(m) { try { if (m && m.src) m.src.disconnect(); } catch (e) {} }
function startCallSpeak(call) {
  if (!call || call.spkT) return;
  call.spkT = setInterval(function () {
    if (S.call !== call || call.ended) { clearInterval(call.spkT); call.spkT = 0; return; }
    if (!call.spkR && call.remote && call.remote.getAudioTracks().length) call.spkR = spkMeter(call.remote);
    if (!call.spkL && call.local) call.spkL = spkMeter(call.local);
    const el = $('call-speak'); if (!el) return;
    const r = spkLevel(call.spkR), l = spkLevel(call.spkL);
    const t = r > 14 ? '对方正在讲话' : (l > 16 ? '你在讲话' : '');
    if (el.textContent !== t) el.textContent = t;
    el.classList.toggle('on', !!t);
  }, 180);
}
function makePC(call) {
  const pc = new RTCPeerConnection({ iceServers: iceServers(), bundlePolicy: 'max-bundle' });
  call.pc = pc;
  pc.onicecandidate = e => { if (e.candidate) queueCand(call, e.candidate.toJSON ? e.candidate.toJSON() : e.candidate); else flushCand(call); };
  pc.ontrack = e => {
    const s = e.streams && e.streams[0];
    if (s) call.remote = s;
    else { if (!call.remote) call.remote = new MediaStream(); if (e.track && !call.remote.getTracks().some(t => t.id === e.track.id)) call.remote.addTrack(e.track); }
    if (e.track) e.track.onunmute = () => { if (S.call === call && !call.ended) attachRemote(call); };   // 对方的声音真正开始到达时再播一次
    attachRemote(call);
  };
  pc.onconnectionstatechange = () => onPcState(call);
  pc.oniceconnectionstatechange = () => onPcState(call);
  call.local.getTracks().forEach(t => pc.addTrack(t, call.local));
  if (call.video) { const lv = $('local-video'); lv.srcObject = call.local; lv.classList.toggle('mirror', call.facing === 'user'); playEl(lv); applyVideoParams(call); $('btn-q').textContent = '画质：' + (QUAL[call.q] || QUAL.hd).name; }
  return pc;
}
function queueCand(call, c) { call.candBuf.push(c); if (!call.candT) call.candT = setTimeout(() => flushCand(call), 100); }
function flushCand(call) {
  clearTimeout(call.candT); call.candT = 0;
  if (!call.candBuf.length || call.ended) return;
  const c = call.candBuf; call.candBuf = []; sig(call.peer, 'ice', { id: call.id, c });
}
async function flushPendingIce(call) {
  const list = S.pendingIce[call.id]; delete S.pendingIce[call.id];
  if (!list || !call.pc) return;
  for (const c of list) { try { await call.pc.addIceCandidate(c); } catch (e) {} }
}
function onPcState(call) {
  if (call.ended || S.call !== call || !call.pc) return;
  let st = call.pc.connectionState; if (!st || st === 'new') st = call.pc.iceConnectionState; if (st === 'completed') st = 'connected';
  if (st === 'connected') {
    clearTimeout(call.dcT);
    if (!call.connected) {
      call.connected = true; call.startedAt = now(); clearTimers(call);
      SFX.backStop(); SFX.connect(); SFX.hold(true); $('call-ui').classList.add('connected'); attachRemote(call);
      startCallSpeak(call);   // “对方正在讲话”
      const tick = () => setCallStatus(fmtDur(Math.floor((now() - call.startedAt) / 1000)));
      tick(); call.clock = setInterval(tick, 500);
      applyCallParams(call); clearInterval(call.statsT); call.statsT = setInterval(() => pollStats(call), 2000); setTimeout(() => pollStats(call), 600);
    }
  } else if (st === 'disconnected') {
    if (call.connected) setCallStatus('网络不稳定，重连中…');
    clearTimeout(call.dcT);
    call.dcT = setTimeout(() => {
      if (S.call !== call || !call.pc) return;
      const s2 = call.pc.connectionState || call.pc.iceConnectionState;
      if (s2 === 'disconnected' || s2 === 'failed') endCall('网络中断', true);
    }, 8000);
  } else if (st === 'failed') {
    endCall(call.connected ? '网络中断' : '连接失败：网络受限（可能需要配置 TURN 服务器）', true);
  }
}

/* 通话权限：先查状态，被拒绝时给可操作的指引，而不是一句“权限不足” */
async function mediaPerm() {
  const out = { mic: 'unknown', cam: 'unknown' };
  try {
    if (!navigator.permissions || !navigator.permissions.query) return out;
    const m = await navigator.permissions.query({ name: 'microphone' });
    out.mic = m.state;
    const q = await navigator.permissions.query({ name: 'camera' });
    out.cam = q.state;
  } catch (e) {}
  return out;
}

    await flushPendingIce(call);
    const ans = await pc.createAnswer(); await pc.setLocalDescription(ans);
    const ok = await sig(call.peer, 'accept', { id: call.id, sdp: { type: ans.type, sdp: ans.sdp } });

async function onAccept(p) {
  const call = S.call;
  if (!call || call.id !== p.id || !call.caller || !call.pc || !p.sdp || call.pc.signalingState !== 'have-local-offer') return;
  try {
    clearTimers(call); setCallStatus('连接中…');
    await call.pc.setRemoteDescription(p.sdp);
    await flushPendingIce(call);
    ctimer(call, () => { if (S.call === call && !call.connected) endCall('连接超时：网络受限（可能需要配置 TURN 服务器）', true); }, 25000);
  } catch (e) { logErr('onAccept', e); if (S.call === call) endCall('连接失败', true); }
}
async function onIce(p) {
  const list = Array.isArray(p.c) ? p.c : [];
  const call = S.call;
  if (call && call.id === p.id && call.pc && call.pc.remoteDescription) {
    for (const c of list) { try { await call.pc.addIceCandidate(c); } catch (e) {} }
  } else { (S.pendingIce[p.id] = S.pendingIce[p.id] || []).push.apply(S.pendingIce[p.id], list); }
}
function onSig(p) {
  if (!p || typeof p !== 'object' || typeof p.from !== 'string' || typeof p.id !== 'string') return;
  if (GC.on && p.cid && p.cid === GC.cid && typeof p.t === 'string' && p.t.indexOf('g') === 0) { gcOnSig(p); return; }   // 群语音：成员不一定互为好友，不能按好友过滤
  if (!S.friends.includes(p.from)) return;
  if (sigDup(p)) return;
  switch (p.t) {
    case 'invite': return onInvite(p);
    case 'accept': return onAccept(p);
    case 'ice': return onIce(p);
    case 'reject': case 'busy': { const c = S.call; if (c && c.id === p.id) endCall(p.t === 'busy' ? '对方忙线中' : '对方已拒绝', false); return; }
    case 'end': {
      const c = S.call;
      if (c && c.id === p.id) endCall(c.connected ? '通话已结束' : '对方已取消', false);
      else if (S.incoming && S.incoming.id === p.id) { clearIncoming(); toast('对方已取消通话'); }
      delete S.pendingIce[p.id]; return;
    }
  }
}

function endCall(reason, notify) {
  const call = S.call;
  if (!call) { hideCallUI(); return; }
  S.call = null; call.ended = true; wakeOff(); SFX.backStop(); SFX.ringStop(); SFX.hold(false); SFX.hangup();
  clearTimers(call); clearTimeout(call.dcT); clearTimeout(call.candT); clearInterval(call.clock); clearInterval(call.statsT); clearInterval(call.spkT); spkStop(call.spkR); spkStop(call.spkL);
  const dur = call.connected && call.startedAt ? Math.floor((now() - call.startedAt) / 1000) : 0;
  if (notify) sig(call.peer, 'end', { id: call.id });
  try { if (call.pc) { call.pc.onicecandidate = null; call.pc.ontrack = null; call.pc.onconnectionstatechange = null; call.pc.oniceconnectionstatechange = null; call.pc.close(); } } catch (e) {}
  stopStream(call.local); stopStream(call.remote);
  delete S.pendingIce[call.id];
  if (call.caller) sendCallLog(call, dur);
  if (call.caller && !call.connected) firePush('miss', call.peer, call.id);   // 推送过来电的：把通知改成“未接来电”（没推送过的数据库会跳过）
  if (reason) { setCallStatus(reason); clearTimeout(S.callHideT); S.callHideT = setTimeout(() => { if (!S.call) hideCallUI(); }, 2200); }
  else hideCallUI();
}

// 声音走 WebRTC mesh：每个人和其余人各连一条，服务器只记“谁在房间里”，不碰声音。
// 限制：手机锁屏/切后台会断；一个房间最多 4 人。
const GC = { on: false, gid: '', cid: '', local: null, peers: {}, roster: [], muted: false, t: 0, tipT: 0, meters: {}, speaking: {}, speakT: 0 };
const GC_MAX = 4;
function gcErr(e) {
  const m = String((e && (e.message || e)) || '');
  if (/最多 4 人/.test(m)) return '群语音最多 4 人，现在满了';
  if (/不在这个群/.test(m)) return '你已经不在这个群里了';
  if (/未登录/.test(m)) return '登录状态掉了，重新登录再试';
  if (/does not exist|PGRST202|schema cache|Could not find/i.test(m)) return '数据库还没更新，请先在后台执行 sql/group-call-v1.sql';
  if (/row-level security|42501/i.test(m)) return '没有权限（数据库策略拦截）';
  return m || '请重试';
}
function gcSupported() {
  if (typeof RTCPeerConnection === 'undefined') { toast('这个浏览器不支持群语音'); return false; }
  if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { toast('这个浏览器拿不到麦克风'); return false; }
  return true;
}
function gcSig(peer, t, data) {
  return sig(peer, t, Object.assign({ id: GC.cid, cid: GC.cid, gid: GC.gid }, data || {}));
}

function gcClosePeer(acc) {
  const st = GC.peers[acc]; if (!st) return;
  st.closed = true; delete GC.peers[acc];
  if (GC.meters[acc]) { spkStop(GC.meters[acc]); delete GC.meters[acc]; }
  delete GC.speaking[acc];
  try { st.pc.onicecandidate = null; st.pc.ontrack = null; st.pc.onconnectionstatechange = null; st.pc.close(); } catch (e) {}
  const a = document.getElementById('gc-a-' + acc); if (a) { try { a.srcObject = null; a.parentNode.removeChild(a); } catch (e) {} }
  if (GC.on) gcRender();
}
function gcNewPeer(acc) {
  const pc = new RTCPeerConnection({ iceServers: iceServers(), bundlePolicy: 'max-bundle' });
  const st = { acc: acc, pc: pc, closed: false, retry: 0 };
  GC.peers[acc] = st;
  GC.local.getTracks().forEach(function (t) { pc.addTrack(t, GC.local); });
  pc.onicecandidate = function (e) {
    if (!e.candidate || st.closed) return;
    gcSig(acc, 'gice', { c: [e.candidate.toJSON ? e.candidate.toJSON() : e.candidate] });
  };
  pc.ontrack = function (e) {
    if (st.closed) return;
    const a = gcAudio(acc);
    const s = (e.streams && e.streams[0]) || new MediaStream([e.track]);
    if (a.srcObject !== s) a.srcObject = s;
    try { a.play(); } catch (err) {}
    if (e.track) e.track.onunmute = function () { try { a.play(); } catch (err) {} };
    if (!GC.meters[acc]) GC.meters[acc] = spkMeter(s);   // 分析对方的音量：谁在讲话
  };
  pc.onconnectionstatechange = function () {
    if (st.closed) return;
    const s2 = pc.connectionState;
    if (s2 === 'failed' || s2 === 'closed') {
      clearTimeout(st.retry);
      st.retry = setTimeout(function () { if (GC.on && GC.peers[acc] === st) gcClosePeer(acc); }, 3000);   // 下一轮 tick 会重连
    }
    if (GC.on) gcRender();
  };
  return st;
}
async function gcOffer(acc) {
  if (!GC.on || !GC.local || GC.peers[acc]) return;
  try {
    const st = gcNewPeer(acc);
    const off = await st.pc.createOffer();
    await st.pc.setLocalDescription(off);
    await gcSig(acc, 'goffer', { sdp: { type: off.type, sdp: off.sdp } });
    clearTimeout(st.retry);
    st.retry = setTimeout(function () {   // 10 秒还没连上：丢掉，下一轮重来
      if (GC.on && GC.peers[acc] === st && st.pc.connectionState !== 'connected') gcClosePeer(acc);
    }, 10000);
  } catch (e) { logErr('gcOffer', e); gcClosePeer(acc); }
}
async function gcOnSig(p) {
  if (!GC.on || !p) return;
  const acc = p.from;
  if (!acc || acc === ME()) return;
  try {
    if (p.t === 'goffer') {
      if (!p.sdp || typeof p.sdp.sdp !== 'string') return;
      const old = GC.peers[acc];
      if (old && (old.pc.connectionState === 'connected' || old.pc.signalingState !== 'stable')) return;
      if (old) gcClosePeer(acc);
      const st = gcNewPeer(acc);
      await st.pc.setRemoteDescription(p.sdp);
      const ans = await st.pc.createAnswer();
      await st.pc.setLocalDescription(ans);
      await gcSig(acc, 'gans', { sdp: { type: ans.type, sdp: ans.sdp } });
    } else if (p.t === 'gans') {
      const st = GC.peers[acc];
      if (st && p.sdp && st.pc.signalingState === 'have-local-offer') await st.pc.setRemoteDescription(p.sdp);
    } else if (p.t === 'gice') {
      const st = GC.peers[acc];
      const list = Array.isArray(p.c) ? p.c : [];
      if (st) for (let i = 0; i < list.length; i++) { try { await st.pc.addIceCandidate(list[i]); } catch (e) {} }
    } else if (p.t === 'gbye') {
      gcClosePeer(acc);
    }
  } catch (e) { logErr('gcSig', e); }
}

}

function gcSyncPeers() {
  if (!GC.on) return;
  const mine = ME() || '';
  const live = {};
  GC.roster.forEach(function (m) { if (m && m.acc && m.acc !== mine) live[m.acc] = 1; });
  Object.keys(GC.peers).forEach(function (acc) { if (!live[acc]) gcClosePeer(acc); });
  Object.keys(live).forEach(function (acc) { if (!GC.peers[acc] && mine && mine < acc) gcOffer(acc); });   // 账号小的发起，避免两边同时发 offer
}
async function gcTick() {
  if (!GC.on || !sb) return;
  const cid = GC.cid, gid = GC.gid;
  try {
    const r = await Promise.all([
      sb.rpc('group_call_touch', { p_cid: cid }),
      sb.rpc('group_call_state', { p_gid: gid })
    ]);
    if (!GC.on || GC.cid !== cid) return;
    const rs = r[1] || {};
    if (rs.error) throw rs.error;
    const st = rs.data && typeof rs.data === 'object' ? rs.data : null;
    if (!st || !Array.isArray(st.members) || !st.members.length) { gcEnd('群语音已结束'); return; }
    if (!st.members.some(function (m) { return m.acc === ME(); })) { gcEnd('你已不在这个语音里'); return; }
    if (st.id !== cid) { gcEnd('群语音已经换了房间'); return; }
    const before = (GC.roster || []).map(function (m) { return m.acc; });
    GC.roster = st.members;
    if (before.length) {
      st.members.forEach(function (m) { if (m.acc !== ME() && before.indexOf(m.acc) < 0) SFX.join(); });
      before.forEach(function (a) { if (a !== ME() && !st.members.some(function (m) { return m.acc === a; })) SFX.leave(); });
    }
    gcRender();
    gcSyncPeers();
  } catch (e) { logErr('gcTick', e); }
}
async function gcStart() {
  if (!isGroupKey(S.conv)) return;
  if (GC.on) { gcShow(); return; }
  if (S.call || S.incoming) { toast('先结束当前通话，再进群语音'); return; }
  if (!gcSupported()) return;
  if (PERM.mic === 'denied') { callPermGuide(false); return; }
  const gid = gidOf(S.conv);
  let stream = null, cid = '';
  try { stream = await gum({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true } }, 30000); }
  catch (e) { toast('拿不到麦克风：' + micError(e)); return; }
  try {
    const r = await sb.rpc('group_call_join', { p_gid: gid });
    if (r.error) throw r.error;
    cid = r.data;
  } catch (e) { stopStream(stream); toast('进不去群语音：' + gcErr(e)); return; }
  refreshPerm();
  GC.on = true; GC.gid = gid; GC.cid = cid; GC.local = stream; GC.peers = {}; GC.roster = []; GC.muted = false;
  GC.meters = {}; GC.speaking = {}; GC.meters['__me'] = spkMeter(stream);
  clearInterval(GC.speakT); GC.speakT = setInterval(gcSpeakTick, 180);
  gcShow(); gcRender();
  SFX.join();
  clearInterval(GC.t); GC.t = setInterval(gcTick, 2500);
  gcTick(); setTimeout(gcTick, 600);
}
function gcEnd(reason) {
  if (!GC.on) { gcHide(); return; }
  const cid = GC.cid;
  clearInterval(GC.speakT); GC.speakT = 0;
  Object.keys(GC.meters).forEach(function (k) { spkStop(GC.meters[k]); });
  GC.meters = {}; GC.speaking = {};
  Object.keys(GC.peers).forEach(function (acc) { try { gcSig(acc, 'gbye', {}); } catch (e) {} gcClosePeer(acc); });
  stopStream(GC.local);
  clearInterval(GC.t); GC.t = 0;
  GC.on = false; GC.gid = ''; GC.cid = ''; GC.local = null; GC.peers = {}; GC.roster = []; GC.muted = false;
  gcHide();
  if (cid && sb) { try { sb.rpc('group_call_leave', { p_cid: cid }).then(function () {}, function () {}); } catch (e) {} }
  if (reason) toast(reason);
}
function gcToggleMute() {
  if (!GC.on) return;
  GC.muted = !GC.muted;
  try { (GC.local.getAudioTracks() || []).forEach(function (t) { t.enabled = !GC.muted; }); } catch (e) {}
  if (sb && GC.cid) { try { sb.rpc('group_call_mute', { p_cid: GC.cid, p_muted: GC.muted }).then(function () {}, function () {}); } catch (e) {} }
  gcRender();
}
function gcShow() { const el = $('gc-ui'); if (el) el.classList.add('active'); gcRender(); }
function gcHide() { const el = $('gc-ui'); if (el) el.classList.remove('active'); }
function gcRender() {
  if (!GC.on) return;
  const el = $('gc-grid'); if (!el) return;
  const mine = ME();
  const others = GC.roster.filter(function (m) { return m.acc !== mine; });
  const meRow = GC.roster.filter(function (m) { return m.acc === mine; });
  const ms = meRow.concat(others);
  el.classList.toggle('n2', ms.length <= 2);
  el.innerHTML = ms.map(function (m) {
    const isMe = m.acc === mine, name = isMe ? '我' : peerName(m.acc);
    const st = GC.peers[m.acc];
    const conn = isMe ? '' : (st && st.pc.connectionState === 'connected' ? '' : '<div class="gcs">连接中…</div>');
    return '<div class="gcp' + (isMe ? ' me' : '') + (GC.speaking[m.acc] ? ' speaking' : '') + '" data-acc="' + esc(m.acc) + '">' + avatarHTML(isMe ? '我' : nickOf(m.acc), avatarOf(m.acc), false, m.acc) +
      '<div class="gcn">' + esc(name) + '</div>' + (GC.speaking[m.acc] ? '<div class="gcv">正在讲话</div>' : '') + (m.muted ? '<div class="gcm">🔇 已静音</div>' : '') + conn + '</div>';
  }).join('');
  const sub = $('gc-sub'); if (sub) { const nm = grpName('g:' + GC.gid); sub.textContent = (nm && nm !== '群聊' ? nm + ' · ' : '') + GC.roster.length + ' 人 · ' + (GC.muted ? '你已静音' : '麦克风开着') + (GC.roster.length >= GC_MAX ? ' · 已满' : ''); }
  const mb = $('gc-mute'); if (mb) { mb.textContent = GC.muted ? '取消静音' : '静音'; mb.classList.toggle('off', GC.muted); }
}
async function gcTipPoll() {
  if (!sb || GC.on || !isGroupKey(S.conv)) { gcTipHide(); return; }
  const gid = gidOf(S.conv);
  try {
    const { data, error } = await sb.rpc('group_call_state', { p_gid: gid });
    if (error) { gcTipHide(); return; }
    if (!isGroupKey(S.conv) || gidOf(S.conv) !== gid) return;
    const st = data && typeof data === 'object' ? data : null;
    if (!st || !Array.isArray(st.members) || !st.members.length) { gcTipHide(); return; }
    const n = st.members.length;
    const inIt = st.members.some(function (m) { return m.acc === ME(); });
    const el = $('gc-tip');
    el.innerHTML = '<span class="gc-dot"></span><b>语音中 · ' + n + ' 人</b><span>' + (inIt ? '回到语音' : '加入') + '</span>';
    el.classList.add('show');
  } catch (e) { gcTipHide(); }
}
function gcTipHide() { const el = $('gc-tip'); if (el) { el.classList.remove('show'); el.innerHTML = ''; } }
