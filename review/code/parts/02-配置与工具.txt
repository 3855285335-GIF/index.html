/* ---- 消息缓存改存 IndexedDB：localStorage 只有 5MB，聊天记录多了会满 ---- */
const IDB = { p: null, bad: false };
function idbOpen() {
  if (IDB.p) return IDB.p;
  IDB.p = new Promise(function (res, rej) {
    if (!window.indexedDB) { IDB.bad = true; rej(new Error('no indexeddb')); return; }
    let r = null;
    try { r = indexedDB.open('chatter', 1); } catch (e) { IDB.bad = true; rej(e); return; }
    r.onupgradeneeded = function () { const db = r.result; if (!db.objectStoreNames.contains('msgs')) db.createObjectStore('msgs'); };
    r.onsuccess = function () { res(r.result); };
    r.onerror = function () { IDB.bad = true; rej(r.error || new Error('idb')); };
  });
  return IDB.p;
}
async function idbGet(k) {
  try { const db = await idbOpen(); return await new Promise(function (res, rej) { const q = db.transaction('msgs', 'readonly').objectStore('msgs').get(k); q.onsuccess = function () { res(q.result); }; q.onerror = function () { rej(q.error); }; }); }
  catch (e) { IDB.bad = true; return null; }
}
async function idbSet(k, v) {
  try {
    const db = await idbOpen();
    await new Promise(function (res, rej) { const tx = db.transaction('msgs', 'readwrite'); tx.objectStore('msgs').put(v, k); tx.oncomplete = function () { res(); }; tx.onerror = function () { rej(tx.error); }; });
  } catch (e) { IDB.bad = true; throw e; }
}
async function idbDel(k) {
  try {
    const db = await idbOpen();
    await new Promise(function (res, rej) { const tx = db.transaction('msgs', 'readwrite'); tx.objectStore('msgs').delete(k); tx.oncomplete = function () { res(); }; tx.onerror = function () { rej(tx.error); }; });
  } catch (e) { IDB.bad = true; }
}
async function idbClear() {
  try {
    const db = await idbOpen();
    await new Promise(function (res, rej) { const tx = db.transaction('msgs', 'readwrite'); tx.objectStore('msgs').clear(); tx.oncomplete = function () { res(); }; tx.onerror = function () { rej(tx.error); }; });
  } catch (e) { IDB.bad = true; }
}
const MSG = { mem: new Map(), uid: null };
const msgKey = c => (S.uid || '') + ':' + c;
function msgCacheClear() { MSG.mem.clear(); MSG.uid = null; }
async function msgCacheGet(conv) {
  if (!S.uid || !conv) return [];
  if (MSG.uid !== S.uid) { MSG.mem.clear(); MSG.uid = S.uid; }
  const k = msgKey(conv);
  if (MSG.mem.has(k)) return MSG.mem.get(k);
  let list = IDB.bad ? null : await idbGet(k);
  if (!Array.isArray(list)) {
    list = LS.get(mk(S.uid, conv), []) || [];          // 旧版 localStorage 缓存：读一次就搬过去
    if (list.length && !IDB.bad) idbSet(k, list).catch(function () {});
    if (list.length) LS.del(mk(S.uid, conv));
  }
  MSG.mem.set(k, list || []);
  return list || [];
}
function msgCacheSave(conv, list) {
  if (!S.uid || !conv) return;
  const k = msgKey(conv);
  MSG.mem.set(k, list);
  if (IDB.bad) { try { LS.set(mk(S.uid, conv), list); } catch (e) {} return; }
  idbSet(k, list).catch(function () { try { LS.set(mk(S.uid, conv), list); } catch (e) {} });
  try {                                              // 最多留 80 个会话的缓存，太久的删掉
    let idx = LS.get('ch2mgi:' + S.uid, []) || [];
    idx = idx.filter(function (x) { return x !== conv; }); idx.push(conv);
    const drop = idx.length > 80 ? idx.splice(0, idx.length - 80) : [];
    LS.set('ch2mgi:' + S.uid, idx);
    drop.forEach(function (c) { const kk = msgKey(c); MSG.mem.delete(kk); if (!IDB.bad) idbDel(kk); });
  } catch (e) {}
}
async function msgMigrate() {
  if (!S.uid || IDB.bad) return;
  try {
    const uid = S.uid, pre = 'ch2m:' + uid + ':';
    const keys = [];
    for (let i = 0; i < localStorage.length; i++) { const k = localStorage.key(i); if (k && k.indexOf(pre) === 0) keys.push(k); }
    for (const k of keys) {
      const conv = k.slice(pre.length), old = LS.get(k, []);
      if (Array.isArray(old) && old.length) await idbSet(msgKey(conv), old);
      LS.del(k);
    }
    LS.del('ch2midx:' + uid);
  } catch (e) { IDB.bad = true; }
}

function convPreview(c) {
  const t = parseQ(c.last_body).text; if (t) return t;
  if (!c.last_ts) {
    if (isGroupKey(c.convKey)) { fixPreview(c.convKey); return '群聊'; }
    return '开始聊天吧';
  }
  let list = c.convKey === S.conv ? S.msgs : ((MSG.mem.get(msgKey(c.convKey))) || (LS.get(mk(S.uid, c.convKey), []) || []));
  for (let i = list.length - 1; i >= 0; i--) { const m = list[i]; if (m && !isHidden(m)) { const p = previewOf(m); if (p) return p; } }
  fixPreview(c.convKey);
  return '[消息]';
}
const pvAsked = new Set();
function fixPreview(key) {   // 本机也没有记录：去取这个会话最后一条消息（每个会话每次打开 App 最多取一次）
  if (!sb || !S.me || pvAsked.has(key)) return; pvAsked.add(key);
  sb.from('messages').select('type,body,created_at').eq('conv_key', key).order('created_at', { ascending: false }).limit(1).then(r => {
    const m = r && !r.error && r.data && r.data[0], c = convOf(key);
    if (m && c && !parseQ(c.last_body).text) { c.last_body = previewOf(m); if (!c.last_ts && m.created_at) c.last_ts = m.created_at; sched('list'); }
  }, () => {});
}
