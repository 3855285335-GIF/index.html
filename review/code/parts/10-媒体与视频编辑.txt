  if ((S.stk || []).length >= STK_LIMIT) throw new Error('“我的表情”最多 ' + STK_LIMIT + ' 个，先删掉一些再加');

  try { return await insertSticker(url, kw); }
  catch (e) {   // 记录没存上：把刚传的文件删掉，免得白占空间

async function addToStickers(m) {
  if (S.v6 === false) { toast(NEED_V6); return; }

  if (m.type === 'sticker' && !isStoreUrl(url)) { toast('这是表情库里的表情，在“表情库”里就能找到'); return; }
  if (!isStoreUrl(url)) { toast('这条消息还没发送完成'); return; }
  if (!stkOk) await loadStickers(true);
  if ((S.stk || []).some(s => s.url === url)) { toast('已经在“我的表情”里了'); return; }
  try {
    toast('正在添加到表情…');
    if (url.indexOf('/chat-media/' + ME() + '/') > 0) await insertSticker(url, '');   // 自己发的图：直接用同一个文件，不重复占空间
    else {                                                                          // 对方发的图：复制一份到自己文件夹，对方清理空间也不影响
      const r = await fetch(disp(url)); if (!r.ok) throw new Error('图片读取失败（HTTP ' + r.status + '）');
      const b = await r.blob();
      if (!/^image\//.test(b.type || '')) throw new Error('这个文件不是图片');
      if (b.size > STK_MAX) throw new Error('图片超过 10MB，不能做表情');

    }
    renderEp(); toast('已添加到“我的表情”，长按可以设置搜索关键词');
  } catch (e) { logErr('stk-from', e); toast('添加失败：' + ((e && e.message) || '请重试')); }
}

/* 读视频时长，并截第 0.1 秒的画面做封面（最长边 720 像素的 JPEG）。只用手机里的文件，不联网；视频本身不动 */
function videoProbe(file) {
  return new Promise(res => {
    let done = false, u = '', dur = null;
    const v = document.createElement('video');
    const fin = poster => {
      if (done) return; done = true; clearTimeout(t);
      try { v.removeAttribute('src'); v.load(); } catch (e) {} try { URL.revokeObjectURL(u); } catch (e) {}
      res({ dur, poster: poster || null });
    };
    const t = setTimeout(() => fin(null), 6000);
    const grab = () => {
      try {
        const w = v.videoWidth, h = v.videoHeight; if (!w || !h) { fin(null); return; }
        const k = Math.min(1, 720 / Math.max(w, h)), c = document.createElement('canvas');
        c.width = Math.max(1, Math.round(w * k)); c.height = Math.max(1, Math.round(h * k));
        const g = c.getContext('2d'); g.drawImage(v, 0, 0, c.width, c.height);
        const px = g.getImageData(0, 0, c.width, c.height).data; let lit = 0;   // 截到的是全黑（有些手机没解出画面）就不用
        for (let i = 0; i < px.length; i += 4 * 97) if (px[i] + px[i + 1] + px[i + 2] > 24) { lit = 1; break; }
        if (!lit || !c.toBlob) { fin(null); return; }
        c.toBlob(b => fin(b && b.size > 0 ? b : null), 'image/jpeg', 0.82);
      } catch (e) { fin(null); }
    };
    v.muted = true; v.playsInline = true; v.setAttribute('playsinline', ''); v.setAttribute('webkit-playsinline', ''); v.preload = 'auto';
    v.onloadedmetadata = () => { const d = v.duration; dur = d > 0 && isFinite(d) ? Math.min(86400, Math.round(d)) : null; try { v.currentTime = Math.min(0.1, (d || 1) / 2); } catch (e) { fin(null); } };
    v.onseeked = () => { v.onseeked = null; grab(); };
    v.onerror = () => fin(null);
    try { u = URL.createObjectURL(file); v.src = u; v.load(); } catch (e) { fin(null); }
  });
}

function vpvFmt(s) { s = Math.max(0, Math.round(s)); return Math.floor(s / 60) + ':' + String(s % 60).padStart(2, '0'); }

async function vpvEncode() {
  const v = $('vpv-video');
  if (!v.captureStream) throw new Error('这台设备的浏览器不支持视频编辑，只能原样发送');
  const start = Math.max(0, VPV.start), end = Math.min(VPV.end, VPV.dur || v.duration || 0);
  if (end - start < 0.2) throw new Error('裁剪后太短了');
  const w = v.videoWidth || 720, h = v.videoHeight || 1280;
  const scale = VPV.compress ? Math.min(1, 720 / Math.max(w, h)) : 1;
  const cw = Math.max(2, Math.round(w * scale / 2) * 2), ch = Math.max(2, Math.round(h * scale / 2) * 2);
  const cv = document.createElement('canvas'); cv.width = cw; cv.height = ch;
  const ctx = cv.getContext('2d');
  const mixed = new MediaStream();
  cv.captureStream(30).getVideoTracks().forEach(function (t) { mixed.addTrack(t); });
  try { const vs = v.captureStream(); if (vs) vs.getAudioTracks().forEach(function (t) { mixed.addTrack(t); }); } catch (e) {}
  const mime = (window.MediaRecorder && MediaRecorder.isTypeSupported && MediaRecorder.isTypeSupported('video/webm;codecs=vp8,opus')) ? 'video/webm;codecs=vp8,opus' : 'video/webm';
  const rec = new MediaRecorder(mixed, { mimeType: mime, videoBitsPerSecond: VPV.compress ? 1600000 : 4000000, audioBitsPerSecond: 96000 });
  VPV.rec = rec; VPV.cancelEncode = false;
  const parts = [];
  rec.ondataavailable = function (e) { if (e.data && e.data.size) parts.push(e.data); };
  const stopped = new Promise(function (r) { rec.onstop = r; });
  $('vpv-progress').style.display = 'flex';
  try { v.pause(); v.currentTime = start; } catch (e) {}
  await new Promise(function (r) { const h2 = function () { v.removeEventListener('seeked', h2); r(); }; v.addEventListener('seeked', h2, { once: true }); setTimeout(r, 700); });
  rec.start(120);
  try { await v.play(); } catch (e) {}
  const total = Math.max(0.2, end - start);
  await new Promise(function (resolve, reject) {
    const tick = function () {
      if (VPV.cancelEncode) { try { rec.stop(); } catch (e) {} reject(new Error('已取消')); return; }
      const t = v.currentTime - start;
      $('vpv-bar-i').style.width = Math.max(0, Math.min(100, Math.round(t / total * 100))) + '%';
      $('vpv-ptext').textContent = '处理中 ' + vpvFmt(Math.min(t, total)) + ' / ' + vpvFmt(total);
      if (v.currentTime >= end || v.ended) { try { rec.stop(); } catch (e) {} resolve(); return; }
      try { ctx.drawImage(v, 0, 0, cw, ch); } catch (e) {}
      requestAnimationFrame(tick);
    };
    tick();
  });
  await stopped;
  try { v.pause(); } catch (e) {}
  $('vpv-progress').style.display = 'none';
  if (VPV.cancelEncode) throw new Error('已取消');
  const blob = new Blob(parts, { type: 'video/webm' });
  if (!blob.size) throw new Error('处理失败，请重试');
  return new File([blob], 'clip.webm', { type: 'video/webm' });
}

  } catch (e) {
    VPV.busy = false;
    toast((e && e.message === '已取消') ? '已取消处理' : ('视频处理失败：' + ((e && e.message) || '请重试')));
  }
}
async function sendVideo(file) {
  if (!S.conv || !canSendHere()) return;
  const pr = await videoProbe(file);
  if (!S.conv) return;
  const r = await openVideoPreview(file, pr);
  if (!r || !S.conv) return;
  const f = r.file;
  const ext = extOfName(f.name) || (f.type.indexOf('webm') >= 0 ? 'webm' : extOfType(f.type, 'mp4'));
  const ex = { _blob: f, _ext: ext, _ct: f.type || 'video/mp4' };
  const poster = r.poster || pr.poster;
  if (poster) { ex._poster = poster; try { ex._pu = URL.createObjectURL(poster); } catch (e) {} }
  addLocal(newMsg('video', URL.createObjectURL(f), r.dur || pr.dur, ex));
}
function sendDoc(file) {
  const n = cpSlice(file.name || '文件', 120), lu = URL.createObjectURL(file);
  let ct = file.type || 'application/octet-stream';
  if (/html|xml|javascript|ecmascript|svg/i.test(ct)) ct = 'text/plain';      // 网页类文件按纯文本存，打开时不会被当成网页运行
  addLocal(newMsg('file', JSON.stringify({ u: lu, n, s: file.size }), null, { _blob: file, _ext: extOfName(n) || 'bin', _ct: ct, _fm: { n, s: file.size } }));
}
const kindOf = f => /^image\//.test((f && f.type) || '') ? 'image' : /^video\//.test((f && f.type) || '') ? 'video' : 'file';
async function sendFiles(files) {
  if (!canSendHere()) return;
  const list = Array.from(files || []).filter(f => f && f.size > 0);
  if (!list.length) { if (files && files.length) toast('这个文件是空的'); return; }
  if (list.length > 9) toast('一次最多 9 个，已发送前 9 个');
  if (S.v6 === false && list.some(f => kindOf(f) !== 'image')) await checkSchema();   // 也许刚运行完 setup-v6.sql
  let big = 0, need = 0;
  for (const f of list.slice(0, 9)) {
    if (!S.conv) break;
    const k = kindOf(f);
    if (k !== 'image' && S.v6 === false) { need++; continue; }
    if (f.size > fileMax()) { big++; continue; }
    if (k === 'image') await sendImage(f); else if (k === 'video') await sendVideo(f); else sendDoc(f);
    await new Promise(r => setTimeout(r, 60));
  }
  if (need) toast(NEED_V6);
  else if (big) toast(big + ' 个文件超过 ' + (fileMax() / MB) + 'MB，已跳过（这是 Supabase 免费版单个文件的上限）');
}

/* ---- 保存到手机：先下载好，再调出系统的“存储图像 / 存储视频 / 存储到文件” ---- */
const stampName = t => { const d = new Date(t || now()); return 'Chatter_' + d.getFullYear() + pad2(d.getMonth() + 1) + pad2(d.getDate()) + '_' + pad2(d.getHours()) + pad2(d.getMinutes()) + pad2(d.getSeconds()); };
function dlFile(f) {
  const u = URL.createObjectURL(f), a = document.createElement('a');
  a.href = u; a.download = f.name; a.rel = 'noopener'; document.body.appendChild(a); a.click(); a.remove();
  setTimeout(() => { try { URL.revokeObjectURL(u); } catch (e) {} }, 60000); toast('已下载：' + (f.name || '文件') + '（在浏览器的「下载」文件夹里，Ctrl+J 可查看）');
}
function shareFile(f, second) {
  navigator.share({ files: [f] }).catch(e => {
    if (e && e.name === 'AbortError') return;
    if (e && e.name === 'NotAllowedError' && !second) { openActs('文件准备好了（' + fmtSize(f.size) + '）', [{ label: '保存 / 分享', fn: () => shareFile(f, true) }, { label: '取消', fn: null }]); return; }
    logErr('share', e); dlFile(f);
  });
}
let saving2 = false;
async function saveToPhone(url, name) {
  if (!url || saving2) return;
  if (url.indexOf('blob:') === 0) { toast('还在发送中，等发完再保存'); return; }
  saving2 = true; toast('正在准备…');
  let f = null;
  try {
    const r = await fetch(disp(url)); if (!r.ok) throw new Error('HTTP ' + r.status);
    const b = await r.blob(); f = new File([b], name, { type: b.type || 'application/octet-stream' });
  } catch (e) { logErr('save', e); toast('下载失败：可能已被清理，或网络连不上存储服务'); }
  saving2 = false; if (!f) return;
  let can = false; try { can = !!(navigator.canShare && navigator.share && navigator.canShare({ files: [f] })); } catch (e) {}
  if (can) { shareFile(f); return; }
  const desk = (function () { try { return window.matchMedia('(hover:hover) and (pointer:fine)').matches; } catch (e) { return false; } })();
  if (desk && window.showSaveFilePicker) {
    try {
      const h = await window.showSaveFilePicker({ suggestedName: name });
      const w = await h.createWritable(); await w.write(f); await w.close();
      toast('已保存：' + name); return;
    } catch (e) { if (e && e.name === 'AbortError') return; logErr('save-picker', e); }
  }
  dlFile(f);
}
function saveMsg(m) {
  if (m.type === 'file') { const f = parseFile(m.body); if (f) saveToPhone(f.u, f.n); return; }
  const def = m.type === 'video' ? 'mp4' : 'jpg';
  saveToPhone(m.body, stampName(m.created_at) + '.' + (extOfUrl(m.body) || def));
}

async function syncPins() {
  if (!sb || !S.uid) return;
  try {
    const r = await sb.from('conversations').select('conv_key,pinned_at').eq('owner_acc', ME());
    if (r.error) return;
    const srv = {};
    (r.data || []).forEach(function (x) { if (x.pinned_at) srv[x.conv_key] = Date.parse(x.pinned_at) || 1; });
    pmLoad();
    const flag = 'ch2pinOk:' + (S.uid || '');

function toggleMute(k) {
  pmLoad();
  const on = !mutes[k];
  if (on) mutes[k] = 1; else delete mutes[k];
  LS.set('ch2mute:' + (S.uid || ''), mutes);
  if (on) {
    try {
      const tag = isGroupKey(k) ? 'g:' + gidOf(k) : 'm:' + peerOf(k, ME());
      swReg().then(function (r) { return r && r.getNotifications ? r.getNotifications() : []; }).then(function (list) { (list || []).forEach(function (n) { if ((n.tag || '') === tag) n.close(); }); }, function () {});
    } catch (e) {}
  }
  try {
    if (sb && S.uid) {
      if (on) sb.from('user_mutes').insert({ owner_acc: ME(), conv_key: k }).then(function () {}, function () {});
      else sb.from('user_mutes').delete().eq('owner_acc', ME()).eq('conv_key', k).then(function () {}, function () {});
    }
  } catch (e) {}
  sched('list'); sched('title');

}

/* ---- 文字里的网址变成可以点的链接 ---- */
const URL_RE = /(https?:\/\/[^\s<>"'，。！？、；：（）【】《》「」]+|www\.[a-z0-9-]+\.[^\s<>"'，。！？、；：（）【】《》「」]+)/gi;
function linkify(t) {
  t = String(t == null ? '' : t); let out = '', last = 0;
  String(t).replace(URL_RE, (m, g, off) => {
    const u = m.replace(/[.,!?;:)\]}>]+$/, ''); if (!u) return m;
    out += esc(t.slice(last, off)) + '<a href="' + esc(/^www\./i.test(u) ? 'https://' + u : u) + '" target="_blank" rel="noopener noreferrer">' + esc(u) + '</a>';
    last = off + u.length; return m;
  });
  return out + esc(t.slice(last));
}
function mentionsOf(m) {
  let a = m && m.mentions;
  if (typeof a === 'string') { try { a = JSON.parse(a); } catch (e) { a = null; } }
  return Array.isArray(a) ? a.filter(function (x) { return x && x.acc && x.name; }) : [];
}
function isAtMe(m) { return m.from_acc !== ME() && mentionsOf(m).some(function (x) { return x.acc === ME() || x.acc === '@all'; }); }
function mentionify(text, mentions) {
  let h = linkify(text);
  (Array.isArray(mentions) ? mentions : []).forEach(function (x) {
    if (!x || !x.name) return;
    const tok = esc('@' + x.name);
    h = h.split(tok).join('<span class="at">' + tok + '</span>');
  });
  return h;
}
