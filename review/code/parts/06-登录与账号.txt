    let b = await bootstrap();

    if (!b || !b.me) throw new Error('无法读取账号档案');
    applyBoot(b); showBanner('');

    logErr('boot', e);

    showBanner(e.message && /setup\.sql/.test(e.message) ? e.message : '网络较慢，先显示缓存数据…');
  }
  S.started = true;
  if (linkWait) setTimeout(() => goLink(linkWait), 0);   // 点通知打开的 App：直接进那个聊天

}
async function onAuth(ev, session) {
  if (!session) { if (ev === 'INITIAL_SESSION' || ev === 'SIGNED_OUT') { teardown(); showAuth(); $('login-pass').value = ''; } return; }
  S.token = session.access_token;
  if (ev === 'TOKEN_REFRESHED') { try { sb.realtime.setAuth(session.access_token); } catch (e) {} return; }
  if (S.started && S.uid === session.user.id) return;
  await start(session.user);
}

async function squareJpeg(file, size) {
  let bmp; try { bmp = await createImageBitmap(file, { imageOrientation: 'from-image' }); } catch (e) { bmp = await createImageBitmap(file); }
  const side = Math.min(bmp.width, bmp.height), out = Math.min(size, side);
  const cv = document.createElement('canvas'); cv.width = out; cv.height = out;
  cv.getContext('2d').drawImage(bmp, (bmp.width - side) / 2, (bmp.height - side) / 2, side, side, 0, 0, out, out); if (bmp.close) bmp.close();
  return await new Promise(r => cv.toBlob(r, 'image/jpeg', 0.9));
}

async function fitJpeg(file, maxW, q) {
  let bmp; try { bmp = await createImageBitmap(file, { imageOrientation: 'from-image' }); } catch (e) { bmp = await createImageBitmap(file); }
  const scale = Math.min(1, maxW / bmp.width);
  const w = Math.max(1, Math.round(bmp.width * scale)), h = Math.max(1, Math.round(bmp.height * scale));
  const cv = document.createElement('canvas'); cv.width = w; cv.height = h;
  cv.getContext('2d').drawImage(bmp, 0, 0, w, h); if (bmp.close) bmp.close();
  return await new Promise(r => cv.toBlob(r, 'image/jpeg', q));
}

// 先预加载图片，加载完再淡入，避免“先空一下再闪出来”；同一张不重复设。
function applyMomentCover() {
  const el = $('mo-cover'), img = $('mo-cover-img'); if (!el || !img) return;
  const u = moCoverURL();
  if (!u) { el.classList.remove('has-cover'); img.style.backgroundImage = ''; moCoverApplied = ''; moCoverTried = ''; return; }
  if (u === moCoverApplied) { el.classList.add('has-cover'); return; }
  if (u === moCoverTried) return;
  moCoverTried = u;
  const done = function () {
    if (moCoverTried !== u || !img) return;
    img.style.backgroundImage = 'url("' + String(u).replace(/"/g, '%22') + '")';
    el.classList.add('has-cover'); moCoverApplied = u;
    try { LS.set(moCoverKey(), u); } catch (e) {}
  };
  const pre = new Image();
  pre.onload = done;
  pre.onerror = function () { if (moCoverTried === u) el.classList.remove('has-cover'); };
  pre.src = u;
  if (pre.complete && pre.naturalWidth) done();
}

});

document.addEventListener('keydown', function (e) {
  if (e.key !== 'Escape') return;
  try {
    if ($('act-sheet').classList.contains('active')) { closeActs(); return; }

function searchMsgs() {
  openForm('搜索聊天记录', [{ ph: '输入关键词，比如“发票”', max: 40 }], async function (vals) {
    const q = String((vals && vals[0]) || '').trim().replace(/[%_\\]/g, '');
    if (!q) return;
    toast('搜索中…');
    try {
      const { data, error } = await sb.from('messages')
        .select('id,conv_key,from_acc,to_acc,body,type,created_at')
        .ilike('body', '%' + q + '%')
        .order('created_at', { ascending: false })
        .limit(40);
      if (error) throw error;
      const rows = (data || []).filter(function (m) { return m && m.type !== 'call' && m.type !== 'recalled'; });
      if (!rows.length) { toast('没有找到相关消息'); return; }
      const items = rows.slice(0, 12).map(function (m) {
        const gp = isGroupKey(m.conv_key);
        const other = m.from_acc === ME() ? m.to_acc : m.from_acc;
        const conv = gp ? m.conv_key : convKeyOf(ME(), other);
        const txt = (parseQ(m.body).text || ('[' + m.type + ']')).replace(/\s+/g, ' ').slice(0, 26);
        const label = (gp ? grpName(m.conv_key) + ' · ' + peerName(m.from_acc) : peerName(other)) + '：' + txt;
        return { label: label, fn: function () { openConv(conv, gp ? conv : other); setTimeout(function () { try { const el = document.querySelector('.msg[data-id="' + m.id + '"]'); if (el) { el.scrollIntoView({ block: 'center' }); el.style.transition = 'background .3s'; el.style.background = 'rgba(255,214,0,.28)'; setTimeout(function () { el.style.background = ''; }, 1800); } } catch (x) {} }, 1400); } };
      });
      items.push({ label: '关闭', danger: true, fn: function () {} });
      openActs('找到 ' + rows.length + ' 条', items);
    } catch (e) { logErr('search', e); toast('搜索失败：' + (e && e.message ? e.message : e)); }
  });
}
