function grpErr(e) {
  const m = String((e && (e.message || e)) || '');
  if (/row-level security|42501/i.test(m)) return '没有权限（数据库策略拦截）';
  if (/duplicate|23505/i.test(m)) return '已经有这个群了';
  if (/invalid input syntax/i.test(m)) return '群 id 格式不对';
  return m || '请重试';
}
function grpCacheSave() { try { LS.set('ch2grp:' + (S.uid || ''), GRP.list.map(function (x) { return { id: x.id, name: x.name, owner_acc: x.owner_acc, avatar_url: x.avatar_url || '', mute_all: !!x.mute_all, join_approval: !!x.join_approval }; })); } catch (e) {} }
async function grpLoad(force) {
  if (!sb || !S.me) return;
  if (!force && now() - GRP.loadAt < 5000) return;
  GRP.loadAt = now();
  try {
    const r = await sb.from('chat_group_members').select('group_id').eq('account_id', ME());
    if (r.error) throw r.error;
    const ids = [];
    (r.data || []).forEach(function (x) { if (x && x.group_id && ids.indexOf(x.group_id) < 0) ids.push(x.group_id); });
    let list = [];
    if (ids.length) {
      const g = await sb.from('chat_groups').select('id,name,owner_acc,avatar_url,mute_all,join_approval,notice,notice_at').in('id', ids);
      if (g.error) throw g.error;
      list = g.data || [];
      try {   // 成员数一次查出来，通讯录里的群聊也显示人数
        const cm = await sb.from('chat_group_members').select('group_id').in('group_id', ids);
        if (!cm.error) { const cnt = {}; (cm.data || []).forEach(function (r) { cnt[r.group_id] = (cnt[r.group_id] || 0) + 1; }); list.forEach(function (x) { GRP.count[x.id] = cnt[x.id] || 0; }); }
      } catch (e) {}
    }
    GRP.list = list; grpCacheSave();
    const set = new Set(ids);
    const keep = S.convs.filter(function (c) { return !isGroupKey(c.convKey) || set.has(gidOf(c.convKey)); });
    if (keep.length !== S.convs.length) S.convs = keep;
    list.forEach(function (x) {
      const c = convOf('g:' + x.id);
      if (c && !parseQ(c.last_body).text) fixPreview(c.convKey);
      if (!GRP.cursor['g:' + x.id]) GRP.cursor['g:' + x.id] = Date.parse(c && c.last_ts || 0) || (now() - 60000);
    });
    sched('list');
    sched('contacts');
    if (S.conv && isGroupKey(S.conv)) { renderChatTitle(); renderGrpNotice(); grpOpenMeta(S.conv); }
  } catch (e) { logErr('grpLoad', e); }
}
async function grpMembers(gid, force) {
  if (!gid) return [];
  if (!force && GRP.members[gid] && now() - (GRP.membersAt[gid] || 0) < 10000) return GRP.members[gid];
  const r = await sb.from('chat_group_members').select('account_id,role,title,title_on,muted_until').eq('group_id', gid);
  if (r.error) throw r.error;
  const ms = r.data || [];
  GRP.members[gid] = ms; GRP.membersAt[gid] = now(); GRP.count[gid] = ms.length;
  if (S.conv === 'g:' + gid) { applyMuteState(); if (S.msgs.length) chatRender(S.stick); }
  sched('contacts');
  const need = ms.map(function (m) { return m.account_id; }).filter(function (a) { return a && !S.profiles[a]; });
  if (need.length) {
    const p = await sb.from('profiles').select('account_id,nickname,avatar_url').in('account_id', need);
    if (p && p.data) p.data.forEach(function (x) { S.profiles[x.account_id] = x; });
    sched('list');
  }
  return ms;
}

async function grpCreate(name, members) {

    if (r.error) throw r.error;
    r = await sb.from('chat_group_members').insert({ group_id: gid, account_id: me, role: 'owner' });
    if (r.error) { try { await sb.from('chat_groups').delete().eq('id', gid); } catch (x) {} throw r.error; }
    GRP.members[gid] = [{ account_id: me, role: 'owner' }];
    GRP.count[gid] = 1;
    let ok = 0, fail = 0;
    for (const a of (members || [])) {
      if (!a || a === me) continue;

      if (rr.error) { fail++; logErr('grpCreate-invite:' + a, rr.error); }
      else { ok++; if (rr.data) firePush('ginv', a, rr.data); }
    }
    toast(ok ? '群已创建，已邀请 ' + ok + ' 人' + (fail ? '（' + fail + ' 人失败）' : '') : '群已创建');
    await grpLoad(true);

  } catch (e) { logErr('grpCreate', e); toast('建群失败：' + grpErr(e)); }

function renderGroupFiles(gid, list) {
  const box = $('gi-files'); if (!box) return;
  if (!list.length) { box.innerHTML = '<div class="gi-file-empty">还没有群文件</div>'; return; }
  const mine = (GRP.members[gid] || []).find(function (m) { return m.account_id === ME(); });
  const g = GRP.list.find(function (x) { return x.id === gid; });
  const canAll = (mine && (mine.role === 'owner' || mine.role === 'admin')) || (g && g.owner_acc === ME());
  box.innerHTML = list.map(function (f) {
    const ext = (String(f.name || '').split('.').pop() || 'FILE').toUpperCase().slice(0, 4);
    const canDel = f.uploader === ME() || canAll;
    return '<div class="gi-file" data-id="' + esc(f.id) + '">' +
      '<div class="gf-ic">' + esc(ext) + '</div>' +
      '<div class="gf-main"><div class="gf-name">' + esc(f.name) + '</div><div class="gf-sub">' + gfSize(f.size) + ' · ' + esc(peerName(f.uploader)) + '</div></div>' +
      '<a class="ie-mini" href="' + esc(gfURL(f.path)) + '" target="_blank" rel="noopener" download>下载</a>' +
      (canDel ? '<button class="gf-del" type="button">删</button>' : '') + '</div>';
  }).join('');
  box.onclick = function (e) {
    const del = e.target.closest('.gf-del'); if (!del) return;
    e.preventDefault();
    deleteGroupFile(gid, del.closest('.gi-file').dataset.id);
  };
}
async function uploadGroupFile(gid, file) {
  if (!file || !gid) return;
  if (file.size > 50 * 1048576) { toast('文件最大 50MB'); return; }
  toast('正在上传 ' + file.name + '…');
  const ext = (String(file.name || '').match(/\.([a-z0-9]{1,6})$/i) || [])[1] || 'bin';
  const path = ME() + '/gf_' + now() + '_' + Math.random().toString(36).slice(2, 7) + '.' + ext.toLowerCase();
  const up = await sb.storage.from('chat-media').upload(path, file, { contentType: file.type || 'application/octet-stream', cacheControl: '31536000' });
  if (up.error) { logErr('gf-up', up.error); toast('上传失败：' + (up.error.message || '请重试')); return; }
  const r = await sb.from('group_files').insert({ group_id: gid, uploader: ME(), name: file.name || ('文件.' + ext), size: file.size, mime: file.type || '', path: path });
  if (r.error) { logErr('gf-ins', r.error); toast('保存失败：' + (r.error.message || '请重试')); return; }
  toast('已上传到群文件');
  loadGroupFiles(gid);
}

function groupMuteState(gid) {
  const g = grpOf('g:' + gid) || {}, mine = (GRP.members[gid] || []).find(function (m) { return m.account_id === ME(); });
  const until = mine && mine.muted_until ? Date.parse(mine.muted_until) : 0;
  const role = mine ? mine.role : '';
  const muted = !!mine && (until > now() || (!!g.mute_all && role !== 'owner' && role !== 'admin'));
  return { muted: muted, all: !!g.mute_all, until: until, role: role };
}
async function refreshMyMuteState(gid) {
  if (!sb || !S.me || !gid) return;
  try {
    const [m, g] = await Promise.all([
      sb.from('chat_group_members').select('role,title,title_on,muted_until').eq('group_id', gid).eq('account_id', ME()).maybeSingle(),
      sb.from('chat_groups').select('mute_all').eq('id', gid).maybeSingle()
    ]);
    if (m.data) {
      const arr = GRP.members[gid] || [];
      const i = arr.findIndex(function (x) { return x.account_id === ME(); });
      const row = Object.assign({ account_id: ME() }, arr[i] || {}, m.data);
      if (i >= 0) arr[i] = row; else arr.push(row);
      GRP.members[gid] = arr;
    }
    if (g.data) { const gr = grpOf('g:' + gid); if (gr) gr.mute_all = !!g.data.mute_all; }
    if (S.conv === 'g:' + gid) applyMuteState();
  } catch (e) { logErr('muteState', e); }
}
function applyMuteState() {
  const tip = $('chat-mute-tip'), bar = $('chat-input-bar'), inp = $('chat-input'), send = $('btn-send');
  if (!S.conv || !isGroupKey(S.conv)) {
    if (tip) tip.classList.remove('show');
    if (bar) bar.classList.remove('muted');
    if (inp) { inp.disabled = false; inp.placeholder = '说点什么…'; }
    if (send) send.disabled = false;
    return;
  }
  const st = groupMuteState(gidOf(S.conv));
  if (!st.muted) {
    if (bar) bar.classList.remove('muted');
    if (inp) { inp.disabled = false; inp.placeholder = '说点什么…'; }
    if (send) send.disabled = false;
    if (tip) {
      if (st.all) { tip.textContent = '全员禁言中（群主/管理员仍可发言）'; tip.classList.add('show'); }
      else tip.classList.remove('show');
    }
    return;
  }
  const d = new Date(st.until || now());
  const when = (d.getMonth() + 1) + '月' + d.getDate() + '日 ' + pad2(d.getHours()) + ':' + pad2(d.getMinutes());
  if (tip) { tip.textContent = st.all ? '全员禁言中，只有群主和管理员可以发言' : '你被禁言到 ' + when + '，暂时不能发言'; tip.classList.add('show'); }
  if (bar) bar.classList.add('muted');
  if (inp) { inp.disabled = true; inp.placeholder = '你已被禁言'; }
  if (send) send.disabled = true;
}
function canSendHere() {
  if (!S.conv || !isGroupKey(S.conv)) return true;
  const st = groupMuteState(gidOf(S.conv));
  if (!st.muted) return true;
  toast(st.all ? '全员禁言中，只有群主/管理员可以发言' : '你已被禁言，暂时不能发言');
  return false;
}
async function grpLeave(key) {
  const k = key || S.conv;
  if (!isGroupKey(k)) return;
  const gid = gidOf(k), me = ME();
  const g = grpOf(k) || {};
  const owner = g.owner_acc === me;
  try {
    if (owner) {
      const r1 = await sb.from('chat_group_members').delete().eq('group_id', gid);
      if (r1.error) logErr('grpLeave-members', r1.error);
      const r2 = await sb.from('chat_groups').delete().eq('id', gid);
      if (r2.error) logErr('grpLeave-group', r2.error);
    } else {
      const r = await sb.from('chat_group_members').delete().eq('group_id', gid).eq('account_id', me);
      if (r.error) throw r.error;
    }
    try { await sb.from('conversations').delete().eq('owner_acc', me).eq('conv_key', k); } catch (e) {}
    S.convs = S.convs.filter(function (c) { return c.convKey !== k; });
    GRP.list = GRP.list.filter(function (x) { return x.id !== gid; }); grpCacheSave();
    delete GRP.members[gid]; delete GRP.count[gid]; delete GRP.cursor[k];
    if (S.conv === k) closeChat();
    sched('list'); updateTitle();
    toast(owner ? '群聊已解散' : '已退出群聊');
  } catch (e) { logErr('grpLeave', e); toast('操作失败：' + grpErr(e)); }
}
async function grpPoll() {
  if (!sb || !S.me || document.hidden) return;
  const keys = S.convs.filter(function (c) { return isGroupKey(c.convKey); }).map(function (c) { return c.convKey; }).slice(0, 50);
  if (!keys.length) return;
  keys.forEach(function (k) {
    if (!GRP.cursor[k]) { const c = convOf(k); GRP.cursor[k] = Date.parse(c && c.last_ts || 0) || (now() - 60000); }
  });
  let since = 0;
  keys.forEach(function (k) { const v = GRP.cursor[k]; if (v > since) since = v; });   // 用最新的游标：旧的群消息交给未读数，别把新消息挤出 200 条上限
  try {
    const { data, error } = await sb.from('messages').select('*')
      .in('conv_key', keys).gt('created_at', new Date(since - 5000).toISOString())
      .order('created_at', { ascending: true }).limit(200);
    if (error) throw error;
    let fresh = 0;
    (data || []).forEach(function (r) {
      const m = rowToMsg(r), t = Date.parse(m.created_at) || 0;
      if (GRP.done.has(m.id)) return;
      GRP.done.add(m.id); if (GRP.done.size > 1600) GRP.done = new Set(Array.from(GRP.done).slice(-800));
      const cur = GRP.cursor[m.conv] || 0;
      if (t && t <= cur) return;                          // 打开 App 前就有的消息：不当作新消息
      if (t > cur) GRP.cursor[m.conv] = t;
      if (m.from_acc === ME()) { if (S.conv === m.conv && !S.msgs.some(function (x) { return x.id === m.id; })) upsertMsg(m); return; }
      if (S.conv === m.conv && S.msgs.some(function (x) { return x.id === m.id; })) return;
      fresh++;
      onIncoming(m, true);                                // 未读数交给数据库的会话行，避免重复 +1
    });
    if (fresh) resyncSoon();
  } catch (e) { logErr('grpPoll', e); }
}
